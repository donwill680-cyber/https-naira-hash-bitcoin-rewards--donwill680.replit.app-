const app = express(); const PORT = process.env.PORT || 3000; const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret'; const ADMIN_EMAIL = (process.env.ADMIN_EMAIL || '').toLowerCase(); const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || 'admin123';
